# Development, test and packaging environment. A superset of the runtime requirements,
# so ``pip install -r requirements-dev.txt`` is all a contributor or a build needs.
#
# pytest and pyinstaller live here rather than in requirements.txt because nothing at
# runtime imports either: keeping them in the runtime file made the packaging lockstep
# demand a test framework and a freezer in the build environment, which is a false
# failure waiting to happen.
#
# pyinstaller is declared because it *is* the build. Its absence from any environment on
# this machine is precisely why the shipped artifact fell a day behind the source tree:
# the packaging script could not be run, so nobody noticed that the file the
# distribution guide told testers to download was the previous build with an empty
# knowledge graph.
-r requirements.txt
pytest==9.1.1
# ``backend/pytest.ini`` sets ``timeout`` and ``timeout_method``. Those are options of
# the pytest-timeout plugin, not of pytest itself: without the plugin a clean venv prints
# "PytestConfigWarning: Unknown config option: timeout" and silently runs the whole
# suite with NO per-test ceiling. That is the exact hang the comment in pytest.ini was
# written to prevent. A plugin referenced by configuration must be declared here.
pytest-timeout==2.4.0
pyinstaller>=6.11,<7
# ``tests/test_api.py::test_training_consent_ledger_and_minimized_export`` reaches the
# owned model trainer through ``app.adaptation._runtime_module()``, which imports
# ``model/train.py``. That module's analysis imports are test-only from the backend's
# point of view (nothing in requirements.txt needs them), but they are not optional for
# the test suite. Versions mirror model/provenance.json so the training ledger test
# exercises the same numerical stack the shipped model was produced with.
pandas==3.0.6
pyarrow==26.0.0
scipy==1.18.1
scikit-learn==1.9.1
threadpoolctl==3.7.0

# Property-based testing for the scheduling and grading maths. Layer 1 uses it to search
# for counterexamples instead of pinning a handful of hand-picked numbers; a failure prints
# the minimal input that reproduces it. Pinned like everything else so a green run is
# reproducible rather than "whatever PyPI served that morning".
hypothesis==6.170.15
# Layer 2 runs the whole application against PostgreSQL as well as SQLite. psycopg is the
# only driver app/db.py accepts for a PostgreSQL URL (`postgresql+psycopg://`), so without
# this the PostgreSQL half of the parity test cannot even connect and the run would skip.
psycopg[binary]==3.3.8
# Dependency and vulnerability scanning for the release gate. Lives in the dev file because
# nothing imports it at runtime, for the same reason pytest does not live in requirements.
pip-audit==2.9.0
# pycryptodome backs the encrypted-backup tests (tests/test_database_backup.py) and is
# declared runtime-optional in requirements-optional.txt: the application detects it and
# degrades to a clear BackupEncryptionError when it is absent, so the desktop bundle ships
# without it. The *test suite*, however, does not skip those ten encryption tests when the
# module is missing -- it fails them -- so a clean `pip install -r requirements-dev.txt`
# (exactly what every CI job runs) could not go green on Python 3.12, the version the image
# ships. Declaring it here keeps the test/build environment complete, as this file's header
# promises, without adding it to the runtime bundle.
pycryptodome==3.23.0
